def digest(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def freeze_protocol():
    path = OUT/'protocol.json'
    if path.exists():
        raise RuntimeError('Protocol already exists; use a new result directory for a new study')
    record = {
        'fixed_before_action_collection_utc':time.strftime('%Y-%m-%dT%H:%M:%SZ',time.gmtime()),
        'question':'Can a low-cost learned gate select a certified lightweight update across unseen topology families?',
        'chronology':'The original 576 cases and their original-method outcomes were previously analysed. New action outcomes have not been inspected. This is a fixed extension protocol, not external preregistration.',
        'development':'19 small checks independent of the core benchmark; two-link examples and random 40-flow networks',
        'core_cases':576, 'core_sizes':[1000,10000], 'target':TARGET,
        'phase1_actions':['ray_reuse','always_refine'], 'repeats':REPEATS,
        'refinement':{'predictor':'lambda0*c0/c; positive floor 1e-12',
                      'corrections':2, 'coordinates':8, 'backtracking_trials':8,
                      'ridge':'1e-8 times maximum dual Hessian diagonal'},
        'features':FEATURES,
        'forbidden_inputs':['updated reference allocations or prices','updated optimal values',
                            'test timings','topology name','case identity','shock label'],
        'learning_target':'log(median refinement time / median reuse time) on cases rejected by the initial certificate',
        'model_candidates':['ridge alpha 1','ridge alpha 10','tree depth 1','tree depth 2','tree depth 3'],
        'tree_min_samples_leaf':20, 'decision_thresholds':[0.,-0.05],
        'inner_score':'mean selected action cost relative to reuse, plus observed feature overhead; equal weight to each validation topology',
        'outer_validation':'Leave one complete topology family out; all anchors, route matrices, sizes, shifts, seeds and shocks from that topology stay together',
        'inner_validation':'Leave one of the remaining five topology families out for model and threshold selection',
        'replay_methods':['ray_warm','ray_reuse','always_refine','heuristic','train_best','learned'],
        'heuristic':'Refine if standard deviation of c/c0 exceeds 1e-4; otherwise reuse pipeline',
        'train_best':'Best fixed action by training-topology mean time ratios, without test outcomes',
        'shift_test':{'anchors':'seed101, both sizes, capacity laws and shifts, across all six graph families',
                      'cases':192, 'seed':2026102307,
                      'shocks':['local_05','local_50','heterogeneous_60','mixed_recovery'],
                      'training':'Outer fold models from core observations only; no shift-test outcomes used'},
        'timing':'Fresh paired replay; all online feature extraction, policy prediction, certificate work, lightweight updates and fallback are timed. Data loading and offline training are separate. Actions rotate deterministically within repetitions.',
        'statistics':'Per-case timing medians. Paired ratios and total time; topology-cluster bootstrap with six clusters, 10000 draws, seed20261023. Six convenience topologies do not support population-wide claims.',
        'correctness':'Only an actual feasible-primal/dual certificate accepts a result. Learned predictions never certify quality. Float64 checks are not interval arithmetic.',
        'sources':{p.name:digest(p) for p in [HERE/'certified_updates.py',HERE/'experiment.py',ROOT/'certified_num.py',ROOT/'results/cases.csv',ROOT/'results/anchors.csv']}}
    path.write_text(json.dumps(record,indent=2)+'\n')
    print('Fixed extension protocol written.',flush=True)


def anchor_groups():
    cases = pd.read_csv(ROOT/'results/cases.csv')
    return cases[cases.split=='core'].groupby('anchor',sort=False)


def load_anchor(key):
    with np.load(ROOT/'results/instances'/(key+'.npz'),allow_pickle=False) as f:
        data = {n:f[n] for n in f.files}
    A = sparse.csr_matrix((data['A_data'],data['A_indices'],data['A_indptr']),shape=tuple(data['A_shape']))
    model = NUM(A,data['w'],data['xi'])
    cache = make_cache(model,data['c0'],data['anchor_x'],data['anchor_lam'])
    return model,cache,data


def check_result(model,c,result,reference_x,reference_lam):
    upper = model.dual(result['lam'],c)[0]
    lower = model.utility(result['x'])
    gap = max(0.,(upper-lower)/model.W)
    assert abs(gap-result['gap_per_weight']) < 1e-9
    overload = model.overload(result['x'],c)
    assert overload < 1e-9
    ref_upper = model.dual(reference_lam,c)[0]
    ref_lower = model.utility(reference_x)
    assert upper >= ref_lower-1e-8*model.W
    assert lower <= ref_upper+1e-8*model.W
    if result['converged']:
        assert gap <= TARGET+1e-10
        assert (ref_lower-lower)/model.W <= TARGET+1e-8
    return {'overload':overload,'verified_gap':gap,
            'regret_lower':max(0.,(ref_lower-lower)/model.W),
            'regret_upper':max(0.,(ref_upper-lower)/model.W)}


def result_row(meta,method,repeat,order,result,check,prep):
    return {**meta,'method':method,'repeat':repeat,'order_index':order,
            **{k:result.get(k,0) for k in FIELDS},**check,
            'amortized_seconds':result['seconds']+prep/4}


def collect():
    assert (OUT/'protocol.json').exists()
    assert not (OUT/'action_runs.csv').exists(), 'Do not overwrite measured observations'
    records=[]; descriptors=[]; states=OUT/'action_states'; states.mkdir(exist_ok=True)
    start=time.perf_counter(); case_index=0
    for index,(key,group) in enumerate(anchor_groups()):
        model,cache,data=load_anchor(key)
        for row in group.itertuples():
            c=data[row.shock+'_c']
            meta={k:getattr(row,k) for k in ['case','anchor','topology','seed','d','capacity_family','xi','shock']}
            meta['dataset']='core'
            query=cache.query(c,data['anchor_x'])
            before=time.perf_counter()
            z=features(model,cache,data['c0'],c,data['anchor_lam'],query)
            feature_seconds=time.perf_counter()-before
            descriptors.append({**meta,**dict(zip(FEATURES,z)),
                                'precheck_gap':query['gap_per_weight'],
                                'feature_seconds':feature_seconds,
                                'cache_preparation_seconds':cache.full_preparation_seconds})
            arrays={}
            for repeat in range(REPEATS):
                methods=['ray_reuse','always_refine']
                if (case_index+repeat)%2: methods.reverse()
                for order,method in enumerate(methods):
                    result=run_action(model,cache,data['c0'],c,data['anchor_x'],data['anchor_lam'],method)
                    check=check_result(model,c,result,data[row.shock+'_reference_x'],data[row.shock+'_reference_lam'])
                    records.append(result_row(meta,method,repeat,order,result,check,cache.full_preparation_seconds))
                    if repeat==0:
                        arrays[method+'_x']=result['x'];arrays[method+'_lam']=result['lam']
            np.savez_compressed(states/(row.case+'.npz'),**arrays)
            case_index+=1
        pd.DataFrame(records).to_csv(OUT/'action_runs.csv',index=False)
        pd.DataFrame(descriptors).to_csv(OUT/'features.csv',index=False)
        if (index+1)%12==0:
            print(f'Action collection {index+1}/144 anchors, {case_index} cases, {time.perf_counter()-start:.1f}s',flush=True)
    assert len(records)==576*2*REPEATS
    (OUT/'collection_complete.json').write_text(json.dumps({'cases':case_index,'runs':len(records),'seconds':time.perf_counter()-start,'source_sha256':digest(HERE/'certified_updates.py')},indent=2)+'\n')


def prepare_shift_test():
    assert not (OUT/'shift_cases.csv').exists()
    rows=[]; folder=OUT/'shift_instances';folder.mkdir(exist_ok=True)
    start=time.perf_counter()
    for index,(key,group) in enumerate(anchor_groups()):
        first=group.iloc[0]
        if int(first.seed)!=101:continue
        model,cache,data=load_anchor(key)
        rng=np.random.default_rng(np.random.SeedSequence([2026102307,TOPOLOGIES.index(first.topology),int(first.d),int(first.xi*10),int(first.capacity_family=='independent')]))
        updates={}
        for label,factor in [('local_05',.95),('local_50',.5)]:
            c=data['c0'].copy();chosen=rng.choice(model.m,max(1,int(np.ceil(.05*model.m))),replace=False)
            c[chosen]*=factor;updates[label]=c
        updates['heterogeneous_60']=data['c0']*rng.uniform(.4,1.,model.m)
        updates['mixed_recovery']=data['c0']*rng.uniform(.8,1.2,model.m)
        arrays={}
        for shock,c in updates.items():
            ref=solve(model,c,data['anchor_lam'],1e-7,data['anchor_x'],refine=True)
            arrays[shock+'_c']=c;arrays[shock+'_reference_x']=ref['x'];arrays[shock+'_reference_lam']=ref['lam']
            rows.append({'case':key+'_'+shock,'anchor':key,'topology':first.topology,
                         'seed':101,'d':int(first.d),'capacity_family':first.capacity_family,
                         'xi':float(first.xi),'shock':shock,'dataset':'shift',
                         'reference_gap':ref['gap_per_weight'],'reference_converged':ref['converged']})
        np.savez_compressed(folder/(key+'.npz'),**arrays)
        print(f'Shift references {len(rows)}/192 cases, {time.perf_counter()-start:.1f}s',flush=True)
    pd.DataFrame(rows).to_csv(OUT/'shift_cases.csv',index=False)
    assert len(rows)==192


def replay():
    assert not (OUT/'replay_runs.csv').exists(), 'Do not overwrite measured observations'
    models=json.loads((OUT/'fold_models.json').read_text())
    original=pd.read_csv(ROOT/'results/cases.csv');original=original[original.split=='core'].copy();original['dataset']='core'
    shifted=pd.read_csv(OUT/'shift_cases.csv')
    all_cases=pd.concat([original,shifted],ignore_index=True)
    records=[];start=time.perf_counter();case_index=0
    methods=['ray_warm','ray_reuse','always_refine','heuristic','train_best','learned']
    for index,(key,group) in enumerate(all_cases.groupby('anchor',sort=False)):
        model,cache,data=load_anchor(key)
        shiftfile=OUT/'shift_instances'/(key+'.npz')
        shift_data={}
        if shiftfile.exists():
            with np.load(shiftfile,allow_pickle=False) as f:shift_data={k:f[k] for k in f.files}
        for row in group.itertuples():
            source=data if row.dataset=='core' else shift_data
            c=source[row.shock+'_c']
            meta={k:getattr(row,k) for k in ['case','anchor','topology','seed','d','capacity_family','xi','shock','dataset']}
            fold=models[row.topology]
            for repeat in range(REPEATS):
                offset=(case_index+repeat)%len(methods)
                order_methods=methods[offset:]+methods[:offset]
                for order,method in enumerate(order_methods):
                    policy=fold['model'] if method=='learned' else fold['train_best']
                    result=run_action(model,cache,data['c0'],c,data['anchor_x'],data['anchor_lam'],method,policy)
                    check=check_result(model,c,result,source[row.shock+'_reference_x'],source[row.shock+'_reference_lam'])
                    records.append(result_row(meta,method,repeat,order,result,check,cache.full_preparation_seconds))
            case_index+=1
        pd.DataFrame(records).to_csv(OUT/'replay_runs.csv',index=False)
        if (index+1)%12==0:
            print(f'Paired replay {index+1}/144 anchors, {case_index}/768 cases, {time.perf_counter()-start:.1f}s',flush=True)
    assert len(records)==768*len(methods)*REPEATS
    environment={'python':sys.version,'numpy':np.__version__,'scipy':scipy.__version__,
                 'platform':platform.platform(),'timing_repeats':REPEATS,'threads':1,
                 'replay_seconds':time.perf_counter()-start,
                 'source_sha256':{p.name:digest(p) for p in [HERE/'certified_updates.py',HERE/'experiment.py',HERE/'train_policy.py']},
                 'fold_models_sha256':digest(OUT/'fold_models.json'),
                 'scope':'Synthetic computational workloads on six public topology structures; no production trace or population-wide generalisation claim'}
    (OUT/'environment.json').write_text(json.dumps(environment,indent=2)+'\n')


if __name__=='__main__':
    parser=argparse.ArgumentParser();parser.add_argument('stage',choices=['freeze','collect','shift','replay'])
    args=parser.parse_args()
    {'freeze':freeze_protocol,'collect':collect,'shift':prepare_shift_test,'replay':replay}[args.stage]()
